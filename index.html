app.get('/api/get-coze-token', (req, res) => { try { const COZE_APP_ID = process.env.COZE_APP_ID; const COZE_KEY_ID = process.env.COZE_KEY_ID; const COZE_AUD = process.env.COZE_AUD; let privateKeyRaw = process.env.COZE_PRIVATE_KEY; const privateKey = privateKeyRaw.replace(/\\n/g, '\n'); const now = Math.floor(Date.now() / 1000); const payload = { iss: COZE_APP_ID, aud: COZE_AUD, iat: now, exp: now + 3600 }; const token = jwt.sign(payload, privateKey, { algorithm: 'RS256', kid: COZE_KEY_ID }); res.json({ jwt_token: token }); } catch (err) { res.status(500).json({ error: err.message, logid: err.logid || '' }); } });
